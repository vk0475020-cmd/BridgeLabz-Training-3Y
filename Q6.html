// Q6: Employee Management System class Employee { constructor(id, name, department, salary) { this.id = id; this.name = name; this.department = department; this.salary = salary; // monthly salary } getAnnualSalary() { return this.salary * 12; } // Increase monthly salary by the given percentage applyBonus(percent) { this.salary += (this.salary * percent) / 100; } } const employees = [ new Employee(1, "Amit", "Engineering", 60000), new Employee(2, "Sara", "HR", 45000), new Employee(3, "Ravi", "Finance", 52000), new Employee(4, "Neha", "Engineering", 70000), new Employee(5, "John", "Marketing", 48000), ]; console.log("Annual salaries:"); employees.forEach((e) => console.log(`${e.name} (${e.department}): ${e.getAnnualSalary()}`)); // reduce(): sum all annual salaries, starting from 0 const totalPayout = employees.reduce((sum, e) => sum + e.getAnnualSalary(), 0); console.log("Total annual payout:", totalPayout); // Bonus demo employees[0].applyBonus(10); console.log(`\nAfter 10% bonus, ${employees[0].name}'s annual salary: ${employees[0].getAnnualSalary()}`); console.log("New total payout:", employees.reduce((sum, e) => sum + e.getAnnualSalary(), 0));
