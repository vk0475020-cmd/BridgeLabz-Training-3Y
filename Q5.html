// Q5: Movie ticket booking with RegExp validation const rules = { name: { regex: /^[A-Za-z]+(?: [A-Za-z]+)*$/, message: "Name must contain only alphabets." }, email: { regex: /^[^\s@]+@[^\s@]+\.[^\s@]+$/, message: "Enter a valid email address." }, seats: { regex: /^([1-9]|10)$/, message: "Seats must be a number from 1 to 10." }, }; function validateField(fieldId) { const input = document.getElementById(fieldId); const isValid = rules[fieldId].regex.test(input.value.trim()); input.classList.toggle("valid", isValid); input.classList.toggle("invalid", !isValid); document.getElementById(`${fieldId}-error`).textContent = isValid ? "" : rules[fieldId].message; return isValid; } document.getElementById("bookingForm").addEventListener("submit", (event) => { event.preventDefault(); const allValid = Object.keys(rules).map(validateField).every(Boolean); const ticketBox = document.getElementById("ticket"); if (!allValid) { ticketBox.hidden = true; return; } // Store booking info in an object const booking = { name: document.getElementById("name").value.trim(), email: document.getElementById("email").value.trim(), seats: Number(document.getElementById("seats").value), }; ticketBox.hidden = false; ticketBox.innerHTML = `

Ticket Confirmed

Name: ${booking.name}

Email: ${booking.email}

Seats: ${booking.seats}

`; });
