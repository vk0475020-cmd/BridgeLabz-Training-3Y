// Q9: Shopping cart with RegExp coupon validation class Cart { constructor() { this.items = []; } addItem(name, price, quantity) { this.items.push({ name, price, quantity }); } // Sum of price * quantity for every item getTotal() { return this.items.reduce((total, item) => total + item.price * item.quantity, 0); } // Coupon format: SAVE or DISC followed by 1-2 digits (SAVE20, DISC10) // Capture group 2 holds the discount percentage. applyCoupon(code) { const couponRegex = /^(SAVE|DISC)(\d{1,2})$/; const match = couponRegex.exec(code.trim().toUpperCase()); const total = this.getTotal(); if (!match) { console.log(`Invalid coupon "${code}". Use a format like SAVE20 or DISC10.`); return total; } const percent = Number(match[2]); const finalTotal = total - (total * percent) / 100; console.log(`Coupon ${match[0]} applied: ${percent}% off.`); return finalTotal; } } const cart = new Cart(); cart.addItem("Keyboard", 1500, 1); cart.addItem("Mouse", 600, 2); cart.addItem("USB Cable", 250, 3); console.log("Cart total:", cart.getTotal()); console.log("Final total with SAVE20:", cart.applyCoupon("SAVE20")); console.log("Final total with DISC10:", cart.applyCoupon("DISC10")); console.log("Final total with HELLO5:", cart.applyCoupon("HELLO5"));
