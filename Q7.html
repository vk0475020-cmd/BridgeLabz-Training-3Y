// Q7: Login form validation with specific error messages // Each check: a RegExp and the message shown when it does NOT match const usernameChecks = [ { regex: /^.{5,}$/, message: "Username must be at least 5 characters." }, ]; const passwordChecks = [ { regex: /^.{8,}$/, message: "Password must be at least 8 characters." }, { regex: /\d/, message: "Password must include a number." }, { regex: /[A-Z]/, message: "Password must include an uppercase letter." }, { regex: /[a-z]/, message: "Password must include a lowercase letter." }, { regex: /[^A-Za-z0-9]/, message: "Password must include a special character." }, ]; // Returns the list of failed-check messages (empty list = valid) function getErrors(value, checks) { return checks.filter((check) => !check.regex.test(value)).map((check) => check.message); } function showErrors(fieldId, errors) { const input = document.getElementById(fieldId); input.classList.toggle("valid", errors.length === 0); input.classList.toggle("invalid", errors.length > 0); document.getElementById(`${fieldId}-error`).innerHTML = errors.join("
"); } document.getElementById("loginForm").addEventListener("submit", (event) => { event.preventDefault(); const usernameErrors = getErrors(document.getElementById("username").value, usernameChecks); const passwordErrors = getErrors(document.getElementById("password").value, passwordChecks); showErrors("username", usernameErrors); showErrors("password", passwordErrors); const isValid = usernameErrors.length === 0 && passwordErrors.length === 0; document.getElementById("result").textContent = isValid ? "Login successful!" : ""; });
