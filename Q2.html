// Q2: Student registration form validation with RegExp // Each field: its regex and the message shown when it fails const rules = { name: { regex: /^[A-Za-z]+(?: [A-Za-z]+)*$/, message: "Name must contain only alphabets." }, email: { regex: /^[^\s@]+@[^\s@]+\.[^\s@]+$/, message: "Enter a valid email (example@domain.com)." }, phone: { regex: /^\d{10}$/, message: "Phone must be exactly 10 digits." }, // Lookaheads: at least one uppercase, one digit, one special character password: { regex: /^(?=.*[A-Z])(?=.*\d)(?=.*[^A-Za-z0-9]).+$/, message: "Password needs 1 uppercase letter, 1 number and 1 special character.", }, }; // Validate one field, update border colour + error text, return true/false function validateField(fieldId) { const input = document.getElementById(fieldId); const errorBox = document.getElementById(`${fieldId}-error`); const isValid = rules[fieldId].regex.test(input.value.trim()); input.classList.toggle("valid", isValid); // green border input.classList.toggle("invalid", !isValid); // red border errorBox.textContent = isValid ? "" : rules[fieldId].message; return isValid; } // Live validation while typing Object.keys(rules).forEach((id) => { document.getElementById(id).addEventListener("input", () => validateField(id)); }); document.getElementById("studentForm").addEventListener("submit", (event) => { event.preventDefault(); // Validate every field (map first so all errors show, not just the first) const allValid = Object.keys(rules).map(validateField).every(Boolean); document.getElementById("result").textContent = allValid ? "Registration successful!" : ""; });
